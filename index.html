DECLARATIVE_05 // REPRODUCIBLE

NIXOS

CONFIGURE ONCE.
REBUILD FOREVER.

A declarative Linux distribution where your entire system can be described as code, reproduced, rolled back and shared.

flake.nix 
{
  description = "my system";

  inputs.nixpkgs.url =
    "github:NixOS/nixpkgs";

  outputs = { self, nixpkgs }:
    {
      nixosConfigurations.main =
        nixpkgs.lib.nixosSystem {
          system = "x86_64-linux";
        };
    };
}
